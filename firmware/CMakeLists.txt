cmake_minimum_required(VERSION 3.13)
set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/.deps/pico-sdk" CACHE PATH "Pinned Pico SDK")
set(PICO_TINYUSB_PATH "${CMAKE_CURRENT_LIST_DIR}/.deps/tinyusb" CACHE PATH "Pinned TinyUSB")
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
set(PICO_BOARD desk_fidget_counter CACHE STRING "PCB F1 RP2040")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(desk_knob VERSION 6.0.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(desk-knob main.c board_rp2040.c counter.c button.c dashboard.c
  lcd.c lcd_ui.c usb_descriptors.c)
pico_generate_pio_header(desk-knob "${CMAKE_CURRENT_LIST_DIR}/quadrature_encoder.pio")
target_include_directories(desk-knob PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_options(desk-knob PRIVATE -Os -Wall -Wextra -Wno-misleading-indentation)
target_compile_definitions(desk-knob PRIVATE PICO_RP2040_USB_DEVICE_ENUMERATION_FIX=1)
target_link_libraries(desk-knob pico_stdlib pico_unique_id hardware_clocks hardware_i2c
  hardware_spi hardware_pio tinyusb_device)
pico_enable_stdio_usb(desk-knob 0)
pico_enable_stdio_uart(desk-knob 0)
pico_set_program_name(desk-knob "Desk Fidget Counter RP2040")
pico_set_program_version(desk-knob "6.0.0")
pico_add_extra_outputs(desk-knob)
